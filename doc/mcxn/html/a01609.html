MCUXCLEXAMPLE_FUNCTION(mcuxClXofModes_shake_256_streaming_example)
{
/**************************************************************************/
/* Preparation */
/**************************************************************************/
/* Enable ELS */
if(!mcuxClExample_Els_Init(MCUXCLELS_RESET_DO_NOT_CANCEL))
{
return MCUXCLEXAMPLE_STATUS_ERROR;
}
/* Initialize session */
mcuxClSession_Descriptor_t sessionDesc;
mcuxClSession_Handle_t session = &sessionDesc;
/* Allocate and initialize session */
MCUXCLEXAMPLE_ALLOCATE_AND_INITIALIZE_SESSION(session, MCUXCLEXAMPLE_MAX_WA(MCUXCLXOF_MAX_CPU_WA_BUFFER_SIZE, MCUXCLRANDOMMODES_NCINIT_WACPU_SIZE), 0U)
/* Initialize the PRNG */
MCUXCLEXAMPLE_INITIALIZE_PRNG(session);
/**************************************************************************/
/* Xof computation */
/**************************************************************************/
uint8_t output[MCUXCLXOF_EXAMPLE_SHAKE_256_OUTSIZE];
MCUXCLBUFFER_INIT_RW(outputBuf, session, output, sizeof(output));
MCUXCLBUFFER_INIT_RO(dataBuf, session, data, sizeof(data));
MCUXCLBUFFER_INIT_RO(customizationBuf, session, NULL, 0U);
uint32_t context[MCUXCLXOF_SHAKE256_CONTEXT_SIZE_IN_WORDS];
mcuxClXof_Context_t pContext = (mcuxClXof_Context_t) context;
MCUX_CSSL_FP_FUNCTION_CALL_BEGIN(resultInit, tokenInit, mcuxClXof_init(
/* mcuxCLSession_Handle_t session: */ session,
/* mcuxClXof_Context_t context: */ pContext,
/* mcuxClXof_Algo_t algo: */ mcuxClXof_Algorithm_Shake_256,
/* mcuxCl_InputBuffer pCustomization */ customizationBuf,
/* uint32_t customizationSize */ 0U
));
// mcuxClXof_init is a flow-protected function: Check the protection token and the return value
if((MCUX_CSSL_FP_FUNCTION_CALLED(mcuxClXof_init) != tokenInit) || (MCUXCLXOF_STATUS_OK != resultInit))
{
return MCUXCLEXAMPLE_STATUS_ERROR;
}
MCUX_CSSL_FP_FUNCTION_CALL_END();
MCUX_CSSL_FP_FUNCTION_CALL_BEGIN(resultProcess, tokenProcess, mcuxClXof_process(
/* mcuxCLSession_Handle_t session: */ session,
/* mcuxClXof_Context_t context: */ pContext,
/* mcuxCl_InputBuffer pIn: */ dataBuf,
/* uint32_t inSize: */ sizeof(data)
));
// mcuxClXof_process is a flow-protected function: Check the protection token and the return value
if((MCUX_CSSL_FP_FUNCTION_CALLED(mcuxClXof_process) != tokenProcess) || (MCUXCLXOF_STATUS_OK != resultProcess))
{
return MCUXCLEXAMPLE_STATUS_ERROR;
}
MCUX_CSSL_FP_FUNCTION_CALL_END();
MCUX_CSSL_FP_FUNCTION_CALL_BEGIN(resultGen1, tokenGen1, mcuxClXof_generate(
/* mcuxCLSession_Handle_t session: */ session,
/* mcuxClXof_Context_t context: */ pContext,
/* mcuxCl_Buffer_t pOut, */ outputBuf,
/* uint32_t outSize */ outSize - 10U
));
// mcuxClXof_generate is a flow-protected function: Check the protection token and the return value
if((MCUX_CSSL_FP_FUNCTION_CALLED(mcuxClXof_generate) != tokenGen1) || (MCUXCLXOF_STATUS_OK != resultGen1))
{
return MCUXCLEXAMPLE_STATUS_ERROR;
}
MCUX_CSSL_FP_FUNCTION_CALL_END();
MCUXCLBUFFER_UPDATE(outputBuf, outSize - 10U);
MCUX_CSSL_FP_FUNCTION_CALL_BEGIN(resultGen2, tokenGen2, mcuxClXof_generate(
/* mcuxCLSession_Handle_t session: */ session,
/* mcuxClXof_Context_t context: */ pContext,
/* mcuxCl_Buffer_t pOut, */ outputBuf,
/* uint32_t outSize */ 10U
));
// mcuxClXof_generate is a flow-protected function: Check the protection token and the return value
if((MCUX_CSSL_FP_FUNCTION_CALLED(mcuxClXof_generate) != tokenGen2) || (MCUXCLXOF_STATUS_OK != resultGen2))
{
return MCUXCLEXAMPLE_STATUS_ERROR;
}
MCUX_CSSL_FP_FUNCTION_CALL_END();
MCUX_CSSL_FP_FUNCTION_CALL_BEGIN(resultFinish, tokenFinish, mcuxClXof_finish(
/* mcuxCLSession_Handle_t session: */ session,
/* mcuxClXof_Context_t context: */ pContext
));
// mcuxClXof_finish is a flow-protected function: Check the protection token and the return value
if((MCUX_CSSL_FP_FUNCTION_CALLED(mcuxClXof_finish) != tokenFinish) || (MCUXCLXOF_STATUS_OK != resultFinish))
{
return MCUXCLEXAMPLE_STATUS_ERROR;
}
MCUX_CSSL_FP_FUNCTION_CALL_END();
/**************************************************************************/
/* Verification */
/**************************************************************************/
for(size_t i = 0U; i < MCUXCLXOF_EXAMPLE_SHAKE_256_OUTSIZE; i++)
{
if(expectedOutput[i] != output[i]) // Expect that the computed output matches our expected output
{
return MCUXCLEXAMPLE_STATUS_ERROR;
}
}
/**************************************************************************/
/* Session clean-up */
/**************************************************************************/
if(!mcuxClExample_Session_Clean(session))
{
return MCUXCLEXAMPLE_STATUS_ERROR;
}
/* Disable ELS to leave it in a clean state for the next example. */
if(!mcuxClExample_Els_Disable())
{
return MCUXCLEXAMPLE_STATUS_ERROR;
}
return MCUXCLEXAMPLE_STATUS_OK;
}
