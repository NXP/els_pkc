57MCUX_CSSL_FP_FUNCTION_DEF(mcuxClExample_Key_Init_And_Load)
58static inline bool mcuxClExample_Key_Init_And_Load(mcuxClSession_Handle_t pSession,
59 mcuxClKey_Handle_t pKey,
60 mcuxClKey_Type_t type,
61 const uint8_t * pData,
62 uint32_t keyDataLength,
63 mcuxClEls_KeyProp_t * key_properties,
64 uint32_t * dst,
65 uint8_t key_loading_option)
66{
67 /* Init the key. */
68 MCUX_CSSL_FP_FUNCTION_CALL_BEGIN(result, token, mcuxClKey_init(
69 /* mcuxClSession_Handle_t session */ pSession,
70 /* mcuxClKey_Handle_t key */ pKey,
71 /* mcuxClKey_Type_t type */ type,
72 /* const uint8_t * pKeyData */ pData,
73 /* uint32_t keyDataLength */ keyDataLength)
74 );
75
76 if((MCUX_CSSL_FP_FUNCTION_CALLED(mcuxClKey_init) != token) || (MCUXCLKEY_STATUS_OK != result))
77 {
78 return false;
79 }
80 MCUX_CSSL_FP_FUNCTION_CALL_END();
81
82 /* Set the key properties. */
83 MCUX_CSSL_FP_FUNCTION_CALL_BEGIN(result, token, mcuxClKey_setKeyproperties(pKey,
84 key_properties));
85
86 if((MCUX_CSSL_FP_FUNCTION_CALLED(mcuxClKey_setKeyproperties) != token) || (MCUXCLKEY_STATUS_OK != result))
87 {
88 return false;
89 }
90 MCUX_CSSL_FP_FUNCTION_CALL_END();
91
92 if(MCUXCLEXAMPLE_CONST_EXTERNAL_KEY == key_loading_option)
93 {
94 /* load key into destination memory buffer */
95 MCUX_CSSL_FP_FUNCTION_CALL_BEGIN(result, token, mcuxClKey_loadMemory(
96 /* mcuxClSession_Handle_t pSession: */ pSession,
97 /* mcuxClKey_Handle_t key: */ pKey,
98 /* uint32_t * dstData: */ dst));
99
100 if((MCUX_CSSL_FP_FUNCTION_CALLED(mcuxClKey_loadMemory) != token) || (MCUXCLKEY_STATUS_OK != result))
101 {
102 return false;
103 }
104 MCUX_CSSL_FP_FUNCTION_CALL_END();
105 }
106 else
107 {
108 /* load key into destination key slot of coprocessor (key_slot = '*dst') */
109 MCUX_CSSL_FP_FUNCTION_CALL_BEGIN(result, token, mcuxClKey_loadCopro(
110 /* mcuxClSession_Handle_t pSession: */ pSession,
111 /* mcuxClKey_Handle_t key: */ pKey,
112 /* uint32_t dstSlot: */ *dst));
113
114 if((MCUX_CSSL_FP_FUNCTION_CALLED(mcuxClKey_loadCopro) != token) || (MCUXCLKEY_STATUS_OK != result))
115 {
116 return false;
117 }
118 MCUX_CSSL_FP_FUNCTION_CALL_END();
119 }
120 return true;
121}
